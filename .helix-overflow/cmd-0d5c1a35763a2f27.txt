// ---------------------------------------------------------------- geometry helpers
export function box(parent, M, mat, w, h, d, x, y, z, name) {
  const m = new THREE.Mesh(new THREE.BoxGeometry(w, h, d), M[mat] || M);
  m.position.set(x, y, z);
  m.castShadow = true; m.receiveShadow = true;
  if (name) m.name = name;
  parent.add(m);
  return m;
}
